function inferTimingStatusFromLegacyPoints(legacyPoints: unknown): TimingStatus {
  if (legacyPoints === CANTEEN_CONFIG.onTimeBonus || legacyPoints === 5) {
    return 'on-time';
  }
  if (legacyPoints === CANTEEN_CONFIG.latePenalty || legacyPoints === -5) {
    return 'late';
  }
  return 'on-time';
}
